"""Analyze Wyckoff position multiplicities and representative coordinates from CIF files.

Given a CIF file (e.g. generated by SHELX from X-ray diffraction data), this module
determines the Wyckoff position letter of every atom in the structure using the
spglib symmetry dataset, tallies the multiplicity (number of atoms) per Wyckoff
letter, and reports the fractional coordinates of the first atom on each Wyckoff
position rounded to the closest rational with denominator <= 12.

Usage:
    from solution import analyze_wyckoff_position_multiplicities_and_coordinates
    result = analyze_wyckoff_position_multiplicities_and_coordinates(
        "workspace/cif_files/FeS2_mp-226.cif"
    )
"""

from __future__ import annotations

import glob
import json
import os
import sys
from collections import Counter, defaultdict
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from sympy import Rational

# Maximum denominator used when rounding fractional coordinates to rationals.
MAX_DENOMINATOR = 12


def _to_rational_string(value: float) -> str:
    """Round a fractional coordinate to the closest rational with denominator <= 12."""
    return str(Rational(value).limit_denominator(MAX_DENOMINATOR))


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> (dict[str, dict] | dict[str, Any]):
    """Analyze Wyckoff positions of the structure stored in a CIF file.

    Args:
        filepath: Path to the CIF file.

    Returns:
        A dict with two keys:
        - "wyckoff_multiplicity_dict": mapping of Wyckoff letter (e.g. "a") to
          the number of atoms occupying that Wyckoff position in the unit cell.
        - "wyckoff_coordinates_dict": mapping of Wyckoff letter to the fractional
          coordinates [x, y, z] of the first atom on that position, each rounded
          to the closest rational with denominator <= 12 (e.g. "3/8").
    """
    structure = Structure.from_file(filepath)
    dataset = SpacegroupAnalyzer(structure).get_symmetry_dataset()
    if dataset is None:
        return {
            "wyckoff_multiplicity_dict": {},
            "wyckoff_coordinates_dict": {},
        }

    # One Wyckoff letter per atom in the structure (spglib symmetry dataset).
    wyckoff_letters = dataset.wyckoffs

    # Multiplicity = total number of atoms carrying each Wyckoff letter.
    multiplicity_dict = dict(sorted(Counter(wyckoff_letters).items()))

    # Group atoms by Wyckoff letter, preserving file order so the "first atom"
    # of each position is well defined.
    wyckoff_sites: dict[str, list] = defaultdict(list)
    for site, letter in zip(structure, wyckoff_letters):
        wyckoff_sites[letter].append(site)

    coordinates_dict: dict[str, list[str]] = {}
    for letter in sorted(wyckoff_sites):
        first_site = wyckoff_sites[letter][0]
        coordinates_dict[letter] = [
            _to_rational_string(c) for c in first_site.frac_coords
        ]

    return {
        "wyckoff_multiplicity_dict": multiplicity_dict,
        "wyckoff_coordinates_dict": coordinates_dict,
    }


def main(argv: list[str]) -> int:
    """Process one or all CIF files and print the results as JSON."""
    if len(argv) > 1:
        files = argv[1:]
    else:
        default_dir = os.path.join(os.path.dirname(__file__), "..", "cif_files")
        files = sorted(glob.glob(os.path.join(default_dir, "*.cif")))
        if not files:
            print("No CIF files found; pass file paths as arguments.", file=sys.stderr)
            return 1

    for filepath in files:
        result = analyze_wyckoff_position_multiplicities_and_coordinates(filepath)
        print(json.dumps({"file": filepath, **result}, indent=4))
    return 0


if __name__ == "__main__":
    raise SystemExit(main(sys.argv))
